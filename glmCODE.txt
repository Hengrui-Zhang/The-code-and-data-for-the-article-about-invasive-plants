  data   = data.frame(y_RG = y_RG, x_RG = x_RG)

  year_num = seq(min(dat_RG$year_num), max(dat_RG$year_num), length.out = 200)

                   newdata = data.frame(x_RG = log(new_RG$year_num)),
                   type = "link", se.fit = TRUE)

   ## AICc（基于 AICtweedie）
   calc_AICc_tweedie <- function(fm) {
     aic <- tweedie::AICtweedie(fm)
     
     # k：参数个数（含截距）；这里用 length(coef) 已含截距
     k   <- length(coef(fm))
     
     n   <- stats::nobs(fm)
     
     # 防止 n - k - 1 <= 0 时爆炸
     if ((n - k - 1) <= 0) return(NA_real_)
     
     aic + (2 * k * (k + 1)) / (n - k - 1)
   }
   
   ## 计算两个自变量的 VIF（只有两个解释变量时，VIF = 1/(1-r^2)）
   calc_vif_two <- function(x1, x2) {
     r <- suppressWarnings(stats::cor(x1, x2, use = "complete.obs"))
     if (is.na(r)) return(NA_real_)
     r2 <- r^2
     if (r2 >= 1) return(Inf)
     1 / (1 - r2)
   }
   
   ## ✅ 你要的“adj R方”：自由度修正的 deviance-based pseudo-R2
   calc_adj_R2_dev <- function(fm) {
     D_full <- stats::deviance(fm)
     D_null <- fm$null.deviance
     n      <- stats::nobs(fm)
     p      <- length(stats::coef(fm))   # 参数个数（含截距）
     
     # 防止分母为 0 或负数
     if (is.na(D_full) || is.na(D_null) || D_null <= 0) return(NA_real_)
     if ((n - p) <= 0 || (n - 1) <= 0) return(NA_real_)
     
     1 - (D_full / (n - p)) / (D_null / (n - 1))
   }

## 1) AICc（基于 AICtweedie）
##    ✅ 修正：k = length(coef(fm)) 已包含截距，不要再 +1
##    ✅ 增加保护：n - k - 1 <= 0 时返回 NA
## ----------------------------
calc_AICc_tweedie <- function(fm) {
  aic <- tweedie::AICtweedie(fm)
  k   <- length(coef(fm))     # 参数个数（含截距）
  n   <- stats::nobs(fm)

  if ((n - k - 1) <= 0) return(NA_real_)

  aic + (2 * k * (k + 1)) / (n - k - 1)
}

## ----------------------------

##    R2_adj_dev = 1 - (D_full/(n-p)) / (D_null/(n-1))

  if (is.na(D_full) || is.na(D_null) || D_null <= 0) return(NA_real_)

}

## ----------------------------
## 1) AICc（基于 AICtweedie）
##    ✅ 修正：k = length(coef(fm)) 已包含截距，不要再 +1
##    ✅ 增加保护：n - k - 1 <= 0 时返回 NA
## ----------------------------
calc_AICc_tweedie <- function(fm) {
  aic <- tweedie::AICtweedie(fm)
  k   <- length(coef(fm))     # 参数个数（含截距）
  n   <- stats::nobs(fm)

  if ((n - k - 1) <= 0) return(NA_real_)

  aic + (2 * k * (k + 1)) / (n - k - 1)
}

## ----------------------------
## 2) 两个自变量的 VIF（只有两个解释变量时，VIF = 1/(1-r^2)）
## ----------------------------
calc_vif_two <- function(x1, x2) {
  r <- suppressWarnings(stats::cor(x1, x2, use = "complete.obs"))
  if (is.na(r)) return(NA_real_)
  r2 <- r^2
  if (r2 >= 1) return(Inf)
  1 / (1 - r2)
}

## ----------------------------
## 3) ✅ 你要的“adj R方”：自由度修正的 deviance-based pseudo-R2
##    R2_adj_dev = 1 - (D_full/(n-p)) / (D_null/(n-1))
## ----------------------------
calc_adj_R2_dev <- function(fm) {
  D_full <- stats::deviance(fm)
  D_null <- fm$null.deviance
  n      <- stats::nobs(fm)
  p      <- length(stats::coef(fm))  # 参数个数（含截距）

  if (is.na(D_full) || is.na(D_null) || D_null <= 0) return(NA_real_)
  if ((n - p) <= 0 || (n - 1) <= 0) return(NA_real_)

  1 - (D_full / (n - p)) / (D_null / (n - 1))
}

fit_two_predictors <- function(v, data) {

  ## 用这三个变量的 complete cases（保证 VIF / R2 / AICc / adjR2 完全一致）
  dat <- data[, c(y_var, base_var, v)]
  dat <- dat[complete.cases(dat), , drop = FALSE]

  if (nrow(dat) < 10) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_base = NA_real_, VIF_add = NA_real_,
      stringsAsFactors = FALSE
    ))
  }

  ## VIF（两自变量时相同）
  vif_val <- calc_vif_two(dat[[base_var]], dat[[v]])

  ## Tweedie GLM
  fml <- as.formula(paste0("`", y_var, "` ~ `", base_var, "` + `", v, "`"))
  fm <- tryCatch(
    glm(
      fml,
      family = statmod::tweedie(var.power = 1.5, link.power = 0),
      data = dat
    ),
    error = function(e) NULL
  )

  if (is.null(fm)) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_base = vif_val, VIF_add = vif_val,
      stringsAsFactors = FALSE
    ))
  }

  data.frame(
    add_var    = v,
    n          = stats::nobs(fm),
    R2_pseudo  = 1 - fm$deviance / fm$null.deviance,
    Adj_R2_dev = calc_adj_R2_dev(fm),   # ✅ 新增：adj R方
    AICc       = calc_AICc_tweedie(fm),
    VIF_base   = vif_val,
    VIF_add    = vif_val,
    stringsAsFactors = FALSE
  )
}

# 定义自定义排名函数（基于 AICc）
rank_AICc <- function(model) {
  k <- length(coef(model)) + 1  # 参数数量（固定 p 时，phi 是额外参数）
  n <- nobs(model)              # 样本量
  aic <- AICtweedie(model)      # 原始 AIC
  aicc <- aic + (2 * k * (k + 1)) / (n - k - 1)  # 计算 AICc
  return(aicc)
}

# 定义自定义排名函数（基于 AICc）
rank_AICc <- function(model) {
  k <- length(coef(model)) + 1  # 参数数量（固定 p 时，phi 是额外参数）
  n <- nobs(model)              # 样本量
  aic <- AICtweedie(model)      # 原始 AIC
  aicc <- aic + (2 * k * (k + 1)) / (n - k - 1)  # 计算 AICc
  return(aicc)
}

#----------------------------
# AICc（基于 AICtweedie）
# ✅ 修正：k = length(coef(fm)) 已包含截距，不再 +1
# ✅ 防护：n - k - 1 <= 0 返回 NA
#----------------------------
calc_AICc_tweedie <- function(fm) {
  aic <- tweedie::AICtweedie(fm)
  k   <- length(coef(fm))     # 参数个数（含截距）
  n   <- stats::nobs(fm)
  if ((n - k - 1) <= 0) return(NA_real_)
  aic + (2 * k * (k + 1)) / (n - k - 1)
}

#----------------------------
# ✅ 调整“R方”：自由度修正的 deviance-based pseudo-R2
#   R2_adj_dev = 1 - (D_full/(n-p)) / (D_null/(n-1))
#----------------------------
calc_adj_R2_dev <- function(fm) {
  D_full <- stats::deviance(fm)
  D_null <- fm$null.deviance
  n      <- stats::nobs(fm)
  p      <- length(stats::coef(fm))  # 参数个数（含截距）

  if (is.na(D_full) || is.na(D_null) || D_null <= 0) return(NA_real_)
  if ((n - p) <= 0 || (n - 1) <= 0) return(NA_real_)

  1 - (D_full / (n - p)) / (D_null / (n - 1))
}

#----------------------------
# 拟合：y ~ fixed_vars + v，并输出 adjR2/AICc/VIF(每个自变量)
#----------------------------
fit_four_predictors <- function(v, data) {

  all_x <- c(fixed_vars, v)
  use_cols <- c(y_var, all_x)

  # 每个模型用自己的 complete cases
  dat <- data[, use_cols, drop = FALSE]
  dat <- dat[complete.cases(dat), , drop = FALSE]

  # 样本太少跳过（你可以改阈值）
  if (nrow(dat) < 10) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_tot.richness_qiao = NA_real_,
      VIF_PC1 = NA_real_,
      VIF_PC2 = NA_real_,
      VIF_add = NA_real_,
      stringsAsFactors = FALSE
    ))
  }

  # 构造公式（对中文/特殊字符列名安全）
  rhs <- paste(sprintf("`%s`", all_x), collapse = " + ")
  fml <- as.formula(paste0("`", y_var, "` ~ ", rhs))

  fm <- tryCatch(
    glm(
      fml,
      family = statmod::tweedie(var.power = 1.5, link.power = 0),
      data = dat
    ),
    error = function(e) NULL
  )

  if (is.null(fm)) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_tot.richness_qiao = NA_real_,
      VIF_PC1 = NA_real_,
      VIF_PC2 = NA_real_,
      VIF_add = NA_real_,
      stringsAsFactors = FALSE
    ))
  }

  # deviance-based pseudo R2 + 调整版
  R2_pseudo  <- 1 - fm$deviance / fm$null.deviance
  Adj_R2_dev <- calc_adj_R2_dev(fm)

  AICc   <- calc_AICc_tweedie(fm)
  n_used <- stats::nobs(fm)

  # VIF（car::vif 对 glm 可用）
  vif_vec <- tryCatch(car::vif(fm), error = function(e) NA)

  # car::vif 可能返回向量或矩阵（含 GVIF），统一成“每个变量一个数”
  if (is.matrix(vif_vec)) {
    if (all(c("GVIF", "Df") %in% colnames(vif_vec))) {
      vif_num <- vif_vec[, "GVIF"]^(1/(2 * vif_vec[, "Df"]))
      names(vif_num) <- rownames(vif_vec)
      vif_vec <- vif_num
    } else {
      vif_num <- vif_vec[, 1]
      names(vif_num) <- rownames(vif_vec)
      vif_vec <- vif_num
    }
  }

  get_vif <- function(varname) {
    if (is.atomic(vif_vec) && !is.na(vif_vec[1])) {
      if (varname %in% names(vif_vec)) return(as.numeric(vif_vec[[varname]]))
    }
    NA_real_
  }

  data.frame(
    add_var = v,
    n = n_used,
    R2_pseudo = R2_pseudo,
    Adj_R2_dev = Adj_R2_dev,          # ✅ 新增：调整R方
    AICc = AICc,
    VIF_tot.richness_qiao = get_vif("tot.richness_qiao"),
    VIF_PC1 = get_vif("PC1"),
    VIF_PC2 = get_vif("PC2"),
    VIF_add = get_vif(v),
    stringsAsFactors = FALSE
  )
}

## ----------------------------
## 1) AICc（基于 AICtweedie）
##    ✅ 修正：k = length(coef(fm)) 已包含截距，不再 +1
##    ✅ 防护：n - k - 1 <= 0 返回 NA
## ----------------------------
calc_AICc_tweedie <- function(fm) {
  aic <- tweedie::AICtweedie(fm)
  k   <- length(coef(fm))     # 参数个数（含截距）
  n   <- stats::nobs(fm)
  if ((n - k - 1) <= 0) return(NA_real_)
  aic + (2 * k * (k + 1)) / (n - k - 1)
}

## ----------------------------
## 2) ✅ 调整“R方”：自由度修正的 deviance-based pseudo-R2
##    R2_adj_dev = 1 - (D_full/(n-p)) / (D_null/(n-1))
## ----------------------------
calc_adj_R2_dev <- function(fm) {
  D_full <- stats::deviance(fm)
  D_null <- fm$null.deviance
  n      <- stats::nobs(fm)
  p      <- length(stats::coef(fm))  # 参数个数（含截距）

  if (is.na(D_full) || is.na(D_null) || D_null <= 0) return(NA_real_)
  if ((n - p) <= 0 || (n - 1) <= 0) return(NA_real_)

  1 - (D_full / (n - p)) / (D_null / (n - 1))
}

## ----------------------------
## 3) 拟合：y ~ fixed_vars + v，并输出 R2 / AdjR2 / AICc / VIF
## ----------------------------
fit_four_predictors <- function(v, data) {

  all_x <- c(fixed_vars, v)
  use_cols <- c(y_var, all_x)

  # 每个模型用自己的 complete cases
  dat <- data[, use_cols, drop = FALSE]
  dat <- dat[complete.cases(dat), , drop = FALSE]

  if (nrow(dat) < 10) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_tot.richness_qiao = NA_real_,
      VIF_PC1 = NA_real_,
      VIF_PC2 = NA_real_,
      VIF_add = NA_real_,
      stringsAsFactors = FALSE
    ))
  }

  # 构造公式（对中文/特殊字符列名安全）
  rhs <- paste(sprintf("`%s`", all_x), collapse = " + ")
  fml <- as.formula(paste0("`", y_var, "` ~ ", rhs))

  fm <- tryCatch(
    glm(
      fml,
      family = statmod::tweedie(var.power = 1.5, link.power = 0),
      data = dat
    ),
    error = function(e) NULL
  )

  if (is.null(fm)) {
    return(data.frame(
      add_var = v, n = nrow(dat),
      R2_pseudo = NA_real_,
      Adj_R2_dev = NA_real_,
      AICc = NA_real_,
      VIF_tot.richness_qiao = NA_real_,
      VIF_PC1 = NA_real_,
      VIF_PC2 = NA_real_,
      VIF_add = NA_real_,
      stringsAsFactors = FALSE
    ))
  }

  # deviance-based pseudo R2 + 调整版
  R2_pseudo  <- 1 - fm$deviance / fm$null.deviance
  Adj_R2_dev <- calc_adj_R2_dev(fm)

  AICc   <- calc_AICc_tweedie(fm)
  n_used <- stats::nobs(fm)

  # VIF（car::vif 对 glm 可用）
  vif_vec <- tryCatch(car::vif(fm), error = function(e) NA)

  # car::vif 可能返回向量或矩阵（含 GVIF），统一成“每个变量一个数”
  if (is.matrix(vif_vec)) {
    if (all(c("GVIF", "Df") %in% colnames(vif_vec))) {
      vif_num <- vif_vec[, "GVIF"]^(1/(2 * vif_vec[, "Df"]))
      names(vif_num) <- rownames(vif_vec)
      vif_vec <- vif_num
    } else {
      vif_num <- vif_vec[, 1]
      names(vif_num) <- rownames(vif_vec)
      vif_vec <- vif_num
    }
  }

  get_vif <- function(varname) {
    if (is.atomic(vif_vec) && !is.na(vif_vec[1])) {
      if (varname %in% names(vif_vec)) return(as.numeric(vif_vec[[varname]]))
    }
    NA_real_
  }

  data.frame(
    add_var = v,
    n = n_used,
    R2_pseudo = R2_pseudo,
    Adj_R2_dev = Adj_R2_dev,   # ✅ 新增：调整R方
    AICc = AICc,
    VIF_tot.richness_qiao = get_vif("tot.richness_qiao"),
    VIF_PC1 = get_vif("PC1"),
    VIF_PC2 = get_vif("PC2"),
    VIF_add = get_vif(v),
    stringsAsFactors = FALSE
  )
}

  }

  ## 1) 统一样本：只取本次分析用到的列，然后 complete cases
  ##    这样可保证 4 个模型用的是同一批样本（否则 R2 不可比）

  m2  <- glm(f2,  family = fam, data = dat)
  m12 <- glm(f12, family = fam, data = dat)

  ## 5) 普通 deviance-based pseudo-R2（你原来的写法）
  R2D <- function(m, m0) 1 - (m$deviance / m0$deviance)

  r1  <- R2D(m1,  m0)
  r2  <- R2D(m2,  m0)
  r12 <- R2D(m12, m0)

    var.power = var.power,
    link.power = link.power,

    # 未调整版
    R2_X1 = r1,
    R2_X2 = r2,
    R2_X1X2 = r12,
    a_X1_unique = a,
    b_X2_unique = b,

diff_cols <- c(

  "Relative.difference_C_P",
  "Relative.difference_N_P",

# 确保这 8 列是数值（更稳：先转字符再转数值）
dat[diff_cols] <- lapply(dat[diff_cols], function(x) as.numeric(as.character(x)))

# 变成长表
long_df <- dat %>%
  pivot_longer(cols = all_of(diff_cols),
               names_to = "indicator",
               values_to = "delta") %>%
  filter(!is.na(delta))

# —— 1) 逐指标：单样本 t 检验 vs 0，并汇总结果 —— #
tt_summary <- long_df %>%
  group_by(indicator) %>%
  summarise(
    n = sum(!is.na(delta)),
    mean = mean(delta, na.rm = TRUE),
    sd = sd(delta, na.rm = TRUE),
    t = unname(t.test(delta, mu = 0)$statistic),
    df = unname(t.test(delta, mu = 0)$parameter),
    p = unname(t.test(delta, mu = 0)$p.value),
    ci_low = unname(t.test(delta, mu = 0)$conf.int[1]),
    ci_high = unname(t.test(delta, mu = 0)$conf.int[2]),
    .groups = "drop"
  ) %>%
  mutate(
    m = n(),  # 本次多重比较的检验数（这里应为 8）
    p_bonf = p.adjust(p, method = "bonferroni"), # ✅ Bonferroni（自动min到1）
    p_fdr  = p.adjust(p, method = "BH")          # ✅ FDR（Benjamini–Hochberg）
  )

write.csv(tt_summary,
          "ttest_summary_8indicators_vs0_Bonf_FDR.csv",
          row.names = FALSE, fileEncoding = "UTF-8")


# —— 2) 逐指标：画直方图（只保留 0 的红色虚线） —— #
# 你要的风格：theme_classic + 黑色边框；只有一条 0 虚线（红色）
for (v in diff_cols) {
  dfv <- dat %>%
    select(all_of(v)) %>%
    rename(delta = all_of(v)) %>%
    filter(!is.na(delta))
  
  # binwidth 给一个“自动”方案：Freedman–Diaconis（比固定 0.2 更稳）
  bw <- 2 * IQR(dfv$delta, na.rm = TRUE) / (length(dfv$delta)^(1/3))
  if (!is.finite(bw) || bw <= 0) bw <- (max(dfv$delta, na.rm = TRUE) - min(dfv$delta, na.rm = TRUE)) / 10
  if (!is.finite(bw) || bw <= 0) bw <- 0.2
  
  p <- ggplot(dfv, aes(x = delta)) +
    geom_histogram(binwidth = bw, color = "black", linewidth = 0.6, fill = "lightblue") +
    geom_vline(xintercept = 0, linetype = "dashed", linewidth = 0.9, color = "red") +  # ✅ 只要这一条红虚线
    labs(
      x = paste0(v, "（RG − TR）"),
      y = "频率"
    ) +
    theme_classic() +
    theme(
      panel.border = element_rect(colour = "black", linewidth = 1.2, fill = NA),
      axis.ticks.length = unit(0.3, "cm"),
      axis.ticks = element_line(color = "black", linewidth = 1),
      axis.text  = element_text(size = 13, color = "black"),
      axis.title = element_text(size = 15, color = "black")
    )
  p
  # 文件名里避免中文/特殊字符导致保存失败：做一个安全文件名
  safe_nm <- gsub("[^A-Za-z0-9]+", "_", v)
  
  ggsave(
    filename = paste0("hist_", safe_nm, "_vs0_redline.png"),
    plot = p, width = 5, height = 4.5, units = "in",
    dpi = 500, bg = "white"
  )
}
